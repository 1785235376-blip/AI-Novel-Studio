cmake_minimum_required(VERSION 3.24)
project(studio_fixed_whisper_host LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The fixed Whisper host currently supports Windows only")
endif()
set(WHISPER_SOURCE_DIR "" CACHE PATH "Verified offline whisper.cpp 1.8.2 source")
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h" OR NOT EXISTS "${WHISPER_SOURCE_DIR}/examples/server/httplib.h")
  message(FATAL_ERROR "An explicit verified offline whisper.cpp source is required")
endif()
file(READ "${WHISPER_SOURCE_DIR}/CMakeLists.txt" WHISPER_CMAKE)
if(NOT WHISPER_CMAKE MATCHES "project\\(\"whisper.cpp\" VERSION 1\\.8\\.2\\)")
  message(FATAL_ERROR "The fixed host protocol is pinned to whisper.cpp 1.8.2")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(WHISPER_OPENVINO OFF CACHE BOOL "" FORCE)
set(WHISPER_FFMPEG OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/whisper-source")
add_executable(studio-fixed-whisper-host main.cpp)
set_target_properties(studio-fixed-whisper-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/host-bin")
target_compile_features(studio-fixed-whisper-host PRIVATE cxx_std_17)
target_include_directories(studio-fixed-whisper-host PRIVATE "${WHISPER_SOURCE_DIR}/examples/server" "${WHISPER_SOURCE_DIR}/examples")
target_link_libraries(studio-fixed-whisper-host PRIVATE whisper bcrypt ws2_32)
if(MSVC)
  target_compile_options(studio-fixed-whisper-host PRIVATE /utf-8 /W4)
endif()
install(TARGETS studio-fixed-whisper-host RUNTIME DESTINATION .)
